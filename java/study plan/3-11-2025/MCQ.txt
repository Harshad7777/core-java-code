    {    
			int a = 1;
			int b = 2;
			int c = 3;

			b >>= 1;  // b = 2 >> 1 = 1

    }

public class BitwiseNotOperator{
        	public static void main(String[] args) {
                    	int i = 50;
						System.out.print(~i);     // bitwise NOT of 50 → -(50+1) = -51
						System.out.print(",");
						System.out.print(~--i);   // i=49, ~(49)=-(49+1) = -50
						System.out.print(",");
						System.out.print(~++i);   // i=50 again, ~(50)=-(50+1)=-51

        	}
}
